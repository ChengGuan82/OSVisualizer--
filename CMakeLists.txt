cmake_minimum_required(VERSION 3.21)
project(OSVisualizer VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(os_core src/core/Algorithms.cpp)
target_include_directories(os_core PUBLIC src)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE os_core)
  add_test(NAME core_tests COMMAND core_tests)
endif()
option(OSV_BUILD_GUI "Build the Qt desktop application" ON)
if(OSV_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
  add_executable(OSVisualizer WIN32 src/main.cpp src/ui/MainWindow.cpp src/ui/Charts.cpp src/ui/icons.qrc)
  set_target_properties(OSVisualizer PROPERTIES AUTORCC ON)
  target_link_libraries(OSVisualizer PRIVATE os_core Qt6::Widgets)
  if(BUILD_TESTING)
    add_test(NAME gui_smoke COMMAND OSVisualizer --smoke-test)
    set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endif()
  install(TARGETS OSVisualizer RUNTIME DESTINATION bin)
endif()
